126 public void SaveImages(string path, Layout layout, LayoutState state = null)
127 {
128 var extension = Path.GetExtension(path);
129 var name = Path.ChangeExtension(path, null);

133 foreach (var pair in maps)

181 public Dictionary<int, SKBitmap> CreateImages(Layout layout, LayoutState state = null)
182 {
183 Initialize(layout, state);
184
185 var width = TileSize.X * (Padding.Left + Padding.Right + LayoutBounds.Width);
186 var height = TileSize.Y * (Padding.Top + Padding.Bottom + LayoutBounds.Height);
187 var dict = new Dictionary<int, SKBitmap>();
188
189 foreach (var room in Layout.Rooms.Values)
190 {
191 if (!dict.ContainsKey(room.Position.Z))
192 {
193 var map = new SKBitmap(width, height);
194 DrawMap(map, room.Position.Z);
195 dict.Add(room.Position.Z, map);
196 }
197 }
198
199 return dict;
200 }

256 private void DrawMapTiles(SKCanvas image, int z)
257 {
258 var sampling = SKSamplingOptions.Default;
259 var roomTile = CreateSolidTile(ConvertColor(RoomColor));
260
261 foreach (var room in Layout.Rooms.Values)
262 {
263 // If room Z (layer) value is not equal, go to next room.
264 if (room.Position.Z != z)
265 continue;
266
267 var roomState = LayoutState?.RoomStates[room.Id];
268 var cells = room.Template.Cells;
269 var x0 = (room.Position.Y - LayoutBounds.X + Padding.Left) * TileSize.X;
270 var y0 = (room.Position.X - LayoutBounds.Y + Padding.Top) * TileSize.Y;
271 var cellTile = CreateSolidTile(ConvertColor(room.Color));
272
273 for (int i = 0; i < cells.Rows; i++)
274 {
275 for (int j = 0; j < cells.Columns; j++)
276 {
277 var cell = cells[i, j];
278
279 // If cell is empty, go to next cell.
280 if (cell == null)
281 continue;
282
283 var position = new Vector2DInt(i, j);
284 var isCompletelyVisible = roomState == null || roomState.CellIsVisible(position);
285
286 // If room state is defined and is not visible, go to next cell.
287 if (!isCompletelyVisible && !roomState.IsVisible)
288 continue;
289
290 // Calculate draw position
291 var x = TileSize.X * j + x0;
292 var y = TileSize.Y * i + y0;
293 var point = new SKPoint(x, y);
294
295 // Get adjacent cells
296 var north = cells.GetOrDefault(i - 1, j);
297 var south = cells.GetOrDefault(i + 1, j);
298 var west = cells.GetOrDefault(i, j - 1);
299 var east = cells.GetOrDefault(i, j + 1);
300
301 // Get the wall or door tiles
302 var topTile = GetTile(room, cell, null, position, DoorDirection.Top);
303 var bottomTile = GetTile(room, cell, null, position, DoorDirection.Bottom);
304 var northTile = GetTile(room, cell, north, position, DoorDirection.North);
305 var southTile = GetTile(room, cell, south, position, DoorDirection.South);
306 var westTile = GetTile(room, cell, west, position, DoorDirection.West);
307 var eastTile = GetTile(room, cell, east, position, DoorDirection.East);
308
309 // Add cell background fill
310 var tile = isCompletelyVisible ? cellTile : roomTile;
311 image.DrawImage(tile, point, sampling);
312
313 // Superimpose applicable map tiles
314 if (northTile != null)
315 image.DrawImage(northTile, point, sampling);
316 if (southTile != null)
317 image.DrawImage(southTile, point, sampling);
318 if (westTile != null)
319 image.DrawImage(westTile, point, sampling);
320 if (eastTile != null)
321 image.DrawImage(eastTile, point, sampling);
322 if (topTile != null)
323 image.DrawImage(topTile, point, sampling);
324 if (bottomTile != null)
325 image.DrawImage(bottomTile, point, sampling);
326
327 // Draw features only if the cell is completely visible
328 if (isCompletelyVisible)
329 DrawFeatureTiles(image, cell, point);
330 }
331 }
332 }
333 }

372 private SKImage GetTile(Room room, Cell cell, Cell neighbor, Vector2DInt position, DoorDirection direction)
373 {
374 if (Door.ShowDoor(DoorDrawMode, direction) && cell.GetDoor(direction) != null && DoorExists(room, position, direction))
375 return GetMapTile(MapTileType.GetDoorTileType(direction));
376
377 if (neighbor == null)
378 return GetMapTile(MapTileType.GetWallTileType(direction));
